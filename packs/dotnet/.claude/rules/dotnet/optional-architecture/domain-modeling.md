# Domain Modeling

> **Apply only if this project uses DDD-style aggregates; otherwise delete this file.**
> Base types below (`AggregateRoot<TId>`, `Entity<TId>`, `ITenantScopedEntity`, strongly typed ID base records, EF mapping extensions) stand for **the project's equivalent base types**. Use whatever the codebase already defines; don't introduce them if they don't exist.

Guidelines for implementing DDD models for aggregates, entities, and value objects.

## Implementation

1. Create DDD models in the feature's domain folder (e.g., `Features/[Feature]/Domain`), including aggregates, entities, value objects, IDs, repository interfaces, and EF Core configuration—match the existing layout
2. Core DDD concepts:
   - Aggregates are the root of the DDD model and typically map 1:1 to database tables
   - Entities belong to aggregates but have their own identity
   - Value objects are immutable and have no identity
   - Repositories add, get, update, and remove aggregates—never child entities directly
3. Consider storing child entities and value objects as JSON columns on the aggregate to avoid EF Core `.Include()` chains (if the project follows this approach)
4. For aggregates:
   - Use public sealed classes that inherit from the project's aggregate root base type
   - Give each aggregate its own ID type (see `strongly-typed-ids.md` if the project uses strongly typed IDs)
   - Never use navigation properties to other aggregates (e.g., no `User.Organization` or `Order.Customer`)—reference them by ID
   - Use factory methods when creating aggregates; keep constructors private
   - Keep state private or read-only; use methods for state changes and enforcing business rules
   - Prefer immutable properties
   - In multi-tenant projects, make aggregates tenant-scoped by default (including many-to-many aggregates, even if FK constraints ensure isolation)
   - For many-to-many aggregates, consider if cascade delete is needed—use `OnDelete(DeleteBehavior.Cascade)` if so
5. For entities:
   - Use public sealed classes that inherit from the project's entity base type
   - Give each entity its own ID type
   - Use factory methods when creating entities
   - Use private setters or `init` to control state changes
   - Use methods for state changes and enforcing business rules
6. For value objects:
   - Use records to ensure immutability
   - Value objects do not have an ID
7. EF Core configuration:
   - Configure what EF needs to generate SQL and materialize objects: ID conversions, value objects, collections, JSON columns, cascade behavior
   - If migrations are hand-written (no EF tooling), don't configure primitive properties (length, required, indexes); those constraints live in migrations and validators
   - If migrations are generated by EF tooling, the configuration is the schema source of truth—configure lengths, required, and indexes there, following the existing configurations
8. When implementing a new aggregate, start with the minimum required methods and add more as features require, ensuring each maintains aggregate invariants

## Examples

```csharp
// Invoice.cs
public sealed class Invoice : AggregateRoot<InvoiceId> // ✅ DO: Inherit from the project's aggregate root base type
{
    private Invoice(Guid customerId, Address address)
        : base(InvoiceId.NewId())
    {
        CustomerId = customerId;
        Address = address;
        Status = InvoiceStatus.Created;
        InvoiceLines = ImmutableArray<InvoiceLine>.Empty;
    }

    public Guid CustomerId { get; } // ✅ DO: Reference other aggregates by ID

    public InvoiceStatus Status { get; private set; }

    public Address Address { get; private set; }

    public ImmutableArray<InvoiceLine> InvoiceLines { get; private set; } // ✅ DO: Use immutable collections

    public static Invoice Create(Guid customerId, Address address) // ✅ DO: Use factory methods
    {
        return new Invoice(customerId, address);
    }

    public void MarkAsPaid() // ✅ DO: Use intention-revealing methods for mutations
    {
        Status = InvoiceStatus.Paid;
    }

    public void AddInvoiceLine(string description, decimal price)
    {
        var invoiceLine = InvoiceLine.Create(description, price);
        InvoiceLines = InvoiceLines.Add(invoiceLine);
    }
}

public sealed record Address(string Street, string City, string State, string ZipCode); // ✅ DO: Value objects as records

// InvoiceLine.cs
public sealed class InvoiceLine : Entity<InvoiceLineId>
{
    private InvoiceLine(string description, decimal price)
        : base(InvoiceLineId.NewId())
    {
        Description = description;
        Price = price;
    }

    public string Description { get; init; } // ✅ DO: Use init for properties that cannot change

    public decimal Price { get; init; }

    internal static InvoiceLine Create(string description, decimal price)
    {
        return new InvoiceLine(description, price);
    }
}

// InvoiceTypes.cs
public enum InvoiceStatus
{
    Created,
    Paid
}

// InvoiceConfiguration.cs
public sealed class InvoiceConfiguration : IEntityTypeConfiguration<Invoice>
{
    private static readonly JsonSerializerOptions JsonSerializerOptions = JsonSerializerOptions.Default;

    public void Configure(EntityTypeBuilder<Invoice> builder)
    {
        builder.Property(i => i.Id).HasConversion(id => id.Value, value => new InvoiceId(value)); // or the project's ID mapping helper

        builder.OwnsOne(i => i.Address, b => b.ToJson()); // ✅ DO: Map 1:1 value objects with .ToJson()

        // ✅ DO: Map collections with an explicit JSON conversion (add HasColumnType for the provider's JSON type)
        builder.Property(i => i.InvoiceLines)
            .HasConversion(
                v => JsonSerializer.Serialize(v.ToArray(), JsonSerializerOptions),
                v => JsonSerializer.Deserialize<ImmutableArray<InvoiceLine>>(v, JsonSerializerOptions)
            );
    }
}

// ❌ Anti-patterns to avoid
public class BadInvoice : AggregateRoot<InvoiceId> // ❌ Not sealed
{
    public BadInvoice(InvoiceId id, Guid customerId) : base(id) { } // ❌ Public constructor, ID generated outside
    public string CustomerEmail { get; set; } // ❌ Public setters expose mutable state
    public Customer Customer { get; set; } // ❌ Navigation property to another aggregate
    public List<InvoiceLine> InvoiceLines { get; set; } = new(); // ❌ Mutable collection exposed directly
}

public sealed class BadInvoiceConfiguration : IEntityTypeConfiguration<Invoice>
{
    public void Configure(EntityTypeBuilder<Invoice> builder)
    {
        // ❌ DON'T: Configure FK relationships you don't need at runtime; reference by ID instead
        builder.HasOne<Customer>().WithMany().HasForeignKey(i => i.CustomerId);

        // ✅ DO: Configure a relationship when EF must handle cascade delete at runtime
        builder.HasMany<InvoiceAttachment>().WithOne().HasForeignKey(a => a.InvoiceId).OnDelete(DeleteBehavior.Cascade);
    }
}
```
